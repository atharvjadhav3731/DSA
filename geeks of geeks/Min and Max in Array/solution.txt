class Solution
{
    public static ArrayList<Integer> getMinMax(int[] num)
    {
        int max = Integer.MIN_VALUE;
        int min = Integer.MAX_VALUE;

        for(int i = 0; i < num.length; i++)
        {
            if(max < num[i])
            {
                max = num[i];
            }

            if(min > num[i])
            {
                min = num[i];
            }
        }

        ArrayList<Integer> result = new ArrayList<>();
        result.add(min);
        result.add(max);

        return result;
    }
}
